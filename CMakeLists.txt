cmake_minimum_required(VERSION 3.20)
project(pluto_remastered LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Build the Pluto ABI host with MSVC and -A Win32.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BW "${CMAKE_CURRENT_SOURCE_DIR}/vendor/bwapi")
file(GLOB LIB_SOURCES CONFIGURE_DEPENDS "${BW}/BWAPILIB/Source/*.cpp")
file(GLOB CLIENT_SOURCES CONFIGURE_DEPENDS "${BW}/BWAPIClient/Source/*.cpp")
file(GLOB SHARED_SOURCES CONFIGURE_DEPENDS "${BW}/Shared/*.cpp")
add_library(bwapi_client STATIC ${LIB_SOURCES} ${CLIENT_SOURCES} ${SHARED_SOURCES}
  "${BW}/BWAPILIB/UnitCommand.cpp")
target_include_directories(bwapi_client PUBLIC "${BW}/include" PRIVATE
  "${BW}/include/BWAPI/Client" "${BW}/BWAPIClient/Source" "${BW}/Shared")
target_compile_definitions(bwapi_client PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN
  _CRT_SECURE_NO_WARNINGS _SCL_SECURE_NO_WARNINGS)
target_compile_options(bwapi_client PRIVATE /W0 /permissive-)
add_library(minhook STATIC vendor/minhook/src/buffer.c vendor/minhook/src/hook.c
  vendor/minhook/src/trampoline.c vendor/minhook/src/hde/hde32.c)
target_include_directories(minhook PUBLIC vendor/minhook/include)
add_library(command_translation STATIC src/commands.cpp)
target_include_directories(command_translation PUBLIC src)
add_library(localized_messages STATIC src/messages.cpp)
target_include_directories(localized_messages PUBLIC src)
target_compile_options(localized_messages PRIVATE /W4 /permissive- /utf-8)
add_library(pluto-scr SHARED src/bridge.cpp src/snapshot.cpp src/legacy_view.cpp src/overlay.cpp src/command_watch.cpp)
target_link_libraries(pluto-scr PRIVATE bwapi_client minhook command_translation localized_messages bcrypt user32 gdi32)
target_compile_options(pluto-scr PRIVATE /W4 /permissive- /utf-8)
add_executable(scr-loader src/loader.cpp)
target_compile_definitions(scr-loader PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(scr-loader PRIVATE bcrypt)
target_compile_options(scr-loader PRIVATE /W4 /permissive-)
install(TARGETS pluto-scr scr-loader RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_executable(commands-test tests/commands_test.cpp)
  target_link_libraries(commands-test PRIVATE command_translation)
  add_test(NAME command-translation COMMAND commands-test)
  add_executable(session-test tests/session_test.cpp)
  target_include_directories(session-test PRIVATE src)
  add_test(NAME session-admission COMMAND session-test)
  add_executable(team-stats-test tests/team_stats_test.cpp)
  target_include_directories(team-stats-test PRIVATE src)
  add_test(NAME team-stats-aggregation COMMAND team-stats-test)
  add_executable(manual-lock-test tests/manual_lock_test.cpp)
  target_include_directories(manual-lock-test PRIVATE src)
  target_link_libraries(manual-lock-test PRIVATE command_translation)
  add_test(NAME manual-control-lock COMMAND manual-lock-test)
  add_executable(unit-list-test tests/unit_list_test.cpp)
  target_include_directories(unit-list-test PRIVATE src)
  add_test(NAME unit-list-relinking COMMAND unit-list-test)
  add_executable(messages-test tests/messages_test.cpp)
  target_link_libraries(messages-test PRIVATE localized_messages)
  target_compile_options(messages-test PRIVATE /utf-8)
  add_test(NAME korean-messages COMMAND messages-test)
  add_executable(pluto-observation-probe EXCLUDE_FROM_ALL tests/pluto_observation_probe.cpp)
  target_include_directories(pluto-observation-probe PRIVATE src)
  target_compile_definitions(pluto-observation-probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(pluto-observation-probe PRIVATE /W4 /permissive-)
  target_link_libraries(pluto-observation-probe PRIVATE bcrypt)
